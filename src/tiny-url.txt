import java.util.HashMap;
import java.util.Map;

class TinyUrl {

    private static final String BASE_URL = "https://short.ly/";

    private final Map<String, String> urlMap = new HashMap<>();
    private long nextId = 1;

    public String encode(String longUrl) {
        String shortCode = base62Encode(nextId++);
        urlMap.put(shortCode, longUrl);

        return BASE_URL + shortCode;
    }

    public String decode(String shortUrl) {
        String shortCode =
                shortUrl.substring(shortUrl.lastIndexOf("/") + 1);

        return urlMap.get(shortCode);
    }

    public boolean deleteUrl(String shortUrl) {
        String shortCode =
                shortUrl.substring(shortUrl.lastIndexOf("/") + 1);

        return urlMap.remove(shortCode) != null;
    }

    private String base62Encode(long num) {
        // this collison free
        String chars =
                "0123456789abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ";

        StringBuilder sb = new StringBuilder();

        while (num > 0) {
            sb.append(chars.charAt((int) (num % 62)));
            num /= 62;
        }

        return sb.reverse().toString();
