class PriceTrackingService {

    private final TrackedItemRepository trackedItemRepository;
    private final PriceHistoryRepository priceHistoryRepository;
    private final PriceProviderFactory priceProviderFactory;

    public PriceTrackingService(
            TrackedItemRepository trackedItemRepository,
            PriceHistoryRepository priceHistoryRepository,
            PriceProviderFactory priceProviderFactory) {

        this.trackedItemRepository = trackedItemRepository;
        this.priceHistoryRepository = priceHistoryRepository;
        this.priceProviderFactory = priceProviderFactory;
    }

    public void refreshPrice(String trackedItemId) {

        TrackedItem item =
                trackedItemRepository.findById(trackedItemId)
                        .orElseThrow();

        PriceProvider provider =
                priceProviderFactory.getProvider(item.getPlatform());

        PriceDetails latestPrice =
                provider.fetchPrice(item);

        item.setCurrentPrice(latestPrice.getPrice());
        item.setCurrency(latestPrice.getCurrency());
        item.setLastFetchedAt(latestPrice.getFetchedAt());

        trackedItemRepository.save(item);

        PriceHistory history = new PriceHistory(
                item.getId(),
                latestPrice.getPrice(),
                latestPrice.getCurrency(),
                latestPrice.getFetchedAt()
        );

        priceHistoryRepository.save(history);
    }
}
